## Notas de Implementação

### Estrutura de Dados Detalhada

#### Tabela: users
```sql
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  senha_hash VARCHAR(255) NOT NULL,
  nome VARCHAR(255) NOT NULL,
  role VARCHAR(50) NOT NULL CHECK (role IN ('admin', 'contador', 'gerente', 'visualizador')),
  ativo BOOLEAN DEFAULT true,
  criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  atualizado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### Tabela: empresas
```sql
CREATE TABLE empresas (
  id SERIAL PRIMARY KEY,
  cnpj VARCHAR(18) UNIQUE NOT NULL,
  nome VARCHAR(255) NOT NULL,
  responsavel_id INTEGER REFERENCES users(id),
  ativa BOOLEAN DEFAULT true,
  criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  atualizado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### Tabela: usuario_empresa (RLS)
```sql
CREATE TABLE usuario_empresa (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  empresa_id INTEGER NOT NULL REFERENCES empresas(id) ON DELETE CASCADE,
  UNIQUE(usuario_id, empresa_id)
);
```

#### Tabela: impostos_mensal
```sql
CREATE TABLE impostos_mensal (
  id SERIAL PRIMARY KEY,
  empresa_id INTEGER NOT NULL REFERENCES empresas(id) ON DELETE CASCADE,
  mes INTEGER NOT NULL CHECK (mes >= 1 AND mes <= 12),
  ano INTEGER NOT NULL,
  irpj NUMERIC(12, 2),
  csll NUMERIC(12, 2),
  pis NUMERIC(12, 2),
  cofins NUMERIC(12, 2),
  icms NUMERIC(12, 2),
  inss NUMERIC(12, 2),
  fgts NUMERIC(12, 2),
  honorario NUMERIC(12, 2),
  criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  atualizado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(empresa_id, mes, ano)
);
```

#### Tabela: auditoria
```sql
CREATE TABLE auditoria (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER REFERENCES users(id),
  tabela VARCHAR(100) NOT NULL,
  acao VARCHAR(50) NOT NULL CHECK (acao IN ('INSERT', 'UPDATE', 'DELETE')),
  empresa_id INTEGER REFERENCES empresas(id),
  dados_antes JSONB,
  dados_depois JSONB,
  timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

### Índices
```sql
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_empresas_cnpj ON empresas(cnpj);
CREATE INDEX idx_impostos_empresa_id ON impostos_mensal(empresa_id);
CREATE INDEX idx_impostos_mes_ano ON impostos_mensal(mes, ano);
CREATE INDEX idx_usuario_empresa ON usuario_empresa(usuario_id, empresa_id);
CREATE INDEX idx_auditoria_usuario ON auditoria(usuario_id);
```
